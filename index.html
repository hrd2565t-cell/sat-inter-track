function renderAnalytics(data) { if (!data || data.length === 0) return; let totalAppBudget = data.reduce((sum, item) => sum + Number(item.Approved_Budget || 0), 0); let totalUsedBudget = data.reduce((sum, item) => sum + Number(item.Used_Budget || 0), 0); let totalTarget = data.reduce((sum, item) => sum + Number(item.Target_People || 0), 0); let totalPassed = data.reduce((sum, item) => sum + Number(item.Passed_People || 0), 0); let budgetPercent = totalAppBudget > 0 ? ((totalUsedBudget / totalAppBudget) * 100).toFixed(2) : 0; let passPercent = totalTarget > 0 ? ((totalPassed / totalTarget) * 100).toFixed(2) : 0; document.getElementById('analytics-budget-percent').innerText = budgetPercent + '%'; document.getElementById('analytics-people-percent').innerText = passPercent + '%'; let satData = data.filter(i => (i.Main_Category || '').includes('การกีฬาแห่งประเทศไทย')); let assocData = data.filter(i => (i.Main_Category || '').includes('สมาคมกีฬา')); let satUsed = satData.reduce((sum, item) => sum + Number(item.Used_Budget || 0), 0); let assocUsed = assocData.reduce((sum, item) => sum + Number(item.Used_Budget || 0), 0); let satApp = satData.reduce((sum, item) => sum + Number(item.Approved_Budget || 0), 0); let assocApp = assocData.reduce((sum, item) => sum + Number(item.Approved_Budget || 0), 0); let satBudgetPercent = satApp > 0 ? ((satUsed / satApp) * 100).toFixed(2) : 0; let assocBudgetPercent = assocApp > 0 ? ((assocUsed / assocApp) * 100).toFixed(2) : 0; let completedProjects = data.filter(i => i.Operation_Status === 'ดำเนินการเสร็จสิ้นแล้ว').length; // --------------------------------------------------------- // ส่วนที่ 1: สรุปภาพรวมพื้นฐาน (ตัวเดิม) // --------------------------------------------------------- let insightsText = `📌 จากข้อมูลทั้งหมด ${data.length} โครงการ พบว่าภาพรวมมีการเบิกจ่ายงบประมาณไปแล้ว ${budgetPercent}% `; insightsText += `และมีผู้ผ่านการอบรมคิดเป็น ${passPercent}% ของเป้าหมายผู้เข้าร่วมทั้งหมด

`; insightsText += `🏢 เมื่อเทียบสัดส่วนตามหน่วยงาน ฝั่ง กกท. มีการเบิกจ่ายงบประมาณไปแล้ว ${satBudgetPercent}% `; insightsText += `ในขณะที่ ฝั่งสมาคมกีฬาฯ มีการเบิกจ่ายไปแล้ว ${assocBudgetPercent}%

`; insightsText += `✅ ขณะนี้มีโครงการที่ดำเนินการเสร็จสิ้นแล้วจำนวน ${completedProjects} โครงการ จากทั้งหมด ${data.length} โครงการ`; // --------------------------------------------------------- // ส่วนที่ 2: ระบบวิเคราะห์และสร้างคำแนะนำด้านงบประมาณอัตโนมัติ // --------------------------------------------------------- let overBudget = []; let highBurn = []; let lowBurn = []; let surplusTotal = 0; data.forEach(item => { let approved = Number(item.Approved_Budget || 0); let used = Number(item.Used_Budget || 0); let remaining = approved - used; let percentUsed = approved > 0 ? (used / approved) * 100 : 0; let status = item.Operation_Status; let idName = `${item.ID}`; // 1. ตรวจสอบใช้งบเกิน if (used > approved) { overBudget.push(`${idName} (เกินไป ${Math.abs(remaining).toLocaleString()} บาท)`); } // 2. ตรวจสอบความเสี่ยงงบจะหมดก่อนงานเสร็จ (> 90%) else if (percentUsed >= 90 && status === 'อยู่ระหว่างการดำเนินงาน') { highBurn.push(`${idName} (ใช้ไป ${percentUsed.toFixed(1)}%)`); } // 3. ตรวจสอบความล่าช้าในการเบิกจ่าย (< 30%) else if (percentUsed > 0 && percentUsed < 30 && status === 'อยู่ระหว่างการดำเนินงาน') { lowBurn.push(idName); } // 4. คำนวณงบเหลือส่งคืน if (status === 'ดำเนินการเสร็จสิ้นแล้ว' && remaining > 0) { surplusTotal += remaining; } }); // สร้างกล่องข้อความคำแนะนำ (HTML) let adviceHTML = `
`; adviceHTML += `

📑 ข้อเสนอแนะเชิงบริหารงบประมาณ:

    `; // เพิ่มคำแนะนำตามเงื่อนไขที่พบ let hasAdvice = false; if (overBudget.length > 0) { adviceHTML += `
  • 🚨 พบโครงการที่เบิกจ่ายเกินงบอนุมัติ: ได้แก่ ${overBudget.join(', ')} แนะนำให้ตรวจสอบสาเหตุการใช้งบเกินและดำเนินการปรับปรุงตัวเลขบัญชี
    • `; hasAdvice = true; } if (highBurn.length > 0) { adviceHTML += `
  • ⚠️ เฝ้าระวังงบประมาณหมดก่อนปิดโครงการ: พบโครงการ ${highBurn.join(', ')} ที่ยังดำเนินการไม่เสร็จ แต่ใช้งบไปแล้วกว่า 90% แนะนำให้ควบคุมค่าใช้จ่ายในกิจกรรมที่เหลืออย่างรัดกุม
    • `; hasAdvice = true; } if (lowBurn.length > 0) { adviceHTML += `
  • ⏱️ โครงการที่เบิกจ่ายล่าช้า: โครงการ ${lowBurn.join(', ')} มีการเบิกจ่ายต่ำกว่า 30% แนะนำให้ติดตามเร่งรัดการจัดกิจกรรมหรือกระบวนการเบิกจ่ายให้เป็นไปตามแผน
    • `; hasAdvice = true; } if (surplusTotal > 0) { adviceHTML += `
  • 💸 งบประมาณเหลือจ่ายส่งคืน: มีโครงการที่ดำเนินการเสร็จสิ้นและมีเงินเหลือจ่ายรวม ${surplusTotal.toLocaleString()} บาท แนะนำให้รวบรวมเอกสารเพื่อส่งคืนเงินเข้ากองทุนฯ ต่อไป
    • `; hasAdvice = true; } if (!hasAdvice) { adviceHTML += `
  • ✨ ภาพรวมการบริหารงบประมาณอยู่ในเกณฑ์ปกติและเป็นไปตามแผนที่กำหนดไว้
    • `; } adviceHTML += `
`; // นำข้อมูลแสดงออกหน้าจอ (รวมเนื้อหาเดิม + กล่องคำแนะนำใหม่) document.getElementById('smart-insights-text').innerHTML = insightsText + adviceHTML; // --------------------------------------------------------- // ส่วนที่ 3: วาดกราฟ // --------------------------------------------------------- if(window.agencyChartInstance) window.agencyChartInstance.destroy(); const ctxAgency = document.getElementById('agencyCompareChart').getContext('2d'); window.agencyChartInstance = new Chart(ctxAgency, { type: 'bar', data: { labels: ['การกีฬาแห่งประเทศไทย (กกท.)', 'สมาคมกีฬาแห่งประเทศไทย'], datasets: [ { label: 'งบอนุมัติ', data: [satApp, assocApp], backgroundColor: '#3b82f6' }, { label: 'งบใช้จริง', data: [satUsed, assocUsed], backgroundColor: '#10b981' } ] }, options: { responsive: true, maintainAspectRatio: false } }); if(budgetChartInstance) budgetChartInstance.destroy(); const ctxBudget = document.getElementById('budgetCompareChart').getContext('2d'); budgetChartInstance = new Chart(ctxBudget, { type: 'bar', data: { labels: data.map(i => i.Sub_Category ? i.Sub_Category.substring(0, 20) + '...' : 'N/A'), datasets: [ { label: 'งบอนุมัติ', data: data.map(i => Number(i.Approved_Budget || 0)), backgroundColor: '#60a5fa' }, { label: 'งบใช้จริง', data: data.map(i => Number(i.Used_Budget || 0)), backgroundColor: '#34d399' } ] }, options: { responsive: true, maintainAspectRatio: false } }); }
