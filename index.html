const SHEET_ID = '1NYXYr6lCvwwJOHU3gxMdR_AtA6sC96Pkb41c01HxIKY'; const SHEET_NAME = 'Data'; function doGet(e) { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getDisplayValues(); if (data.length <= 1) { return ContentService.createTextOutput(JSON.stringify({ status: "success", data: [] })) .setMimeType(ContentService.MimeType.JSON); } const headers = data[0]; const rows = data.slice(1); const result = rows.map(row => { let obj = {}; headers.forEach((header, index) => { obj[header] = row[index]; }); return obj; }); return ContentService.createTextOutput(JSON.stringify({ status: "success", data: result })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const rawPayload = JSON.parse(e.postData.contents); // แปลงให้เป็น Array เสมอ (รองรับทั้งการกรอก 1 รายการ และอัปโหลดไฟล์หลายรายการ) const payloads = Array.isArray(rawPayload) ? rawPayload : [rawPayload]; const data = sheet.getDataRange().getValues(); let idMap = {}; // เก็บแผนผังรหัส ID และเลขแถว เพื่อเช็กอัปเดตอย่างรวดเร็ว for (let i = 1; i < data.length; i++) { if (data[i][0] && data[i][0] !== "-") { idMap[data[i][0]] = i + 1; } } let newRows = []; let updatedCount = 0; let addedCount = 0; payloads.forEach(payload => { const timestamp = new Date(); const id = payload.projectId || "-"; const rowData = [ id, timestamp, payload.mainCategory || "", payload.subCategory || "", payload.projectActivity || "-", payload.projectName || "-", payload.location || "-", payload.startDate || "-", payload.endDate || "-", payload.targetPeople || 0, payload.passedPeople || 0, payload.approvedBudget || 0, payload.usedBudget || 0, payload.operationStatus || "อยู่ระหว่างการดำเนินงาน", payload.paymentStatus || "อยู่ระหว่างการเบิกจ่าย" ]; if (id !== "-" && idMap[id]) { // อัปเดตข้อมูลเดิม sheet.getRange(idMap[id], 1, 1, rowData.length).setValues([rowData]); updatedCount++; } else { // เพิ่มข้อมูลใหม่ newRows.push(rowData); addedCount++; } }); // เขียนข้อมูลใหม่ทั้งหมดในครั้งเดียว if (newRows.length > 0) { sheet.getRange(sheet.getLastRow() + 1, 1, newRows.length, newRows[0].length).setValues(newRows); } let msg = payloads.length === 1 ? (updatedCount > 0 ? "อัปเดตข้อมูลเดิมเรียบร้อย" : "บันทึกโครงการใหม่เรียบร้อย") : `นำเข้าข้อมูลสำเร็จ: เพิ่มใหม่ ${addedCount} รายการ, อัปเดต ${updatedCount} รายการ`; return ContentService.createTextOutput(JSON.stringify({ status: "success", message: msg })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
