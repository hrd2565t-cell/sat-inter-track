function renderTable(data) { const tbody = document.getElementById('dataTableBody'); if(data.length === 0) { tbody.innerHTML = `ไม่พบข้อมูลที่ตรงกับเงื่อนไข`; return; } tbody.innerHTML = data.map(item => ` ${item.ID || ''}
${item.Sub_Category || ''}
${item.Main_Category || ''}
${item.Project_Activity || '-'}
${item.Project_Name && item.Project_Name !== '-' ? '(' + item.Project_Name + ')' : ''}
${item.Location || '-'}
${item.Start_Date || '-'}
ถึง
${item.End_Date || '-'}
เป้าหมาย: ${item.Target_People || 0}
ผ่าน: ${item.Passed_People || 0}
ใช้ไป: ${Number(item.Used_Budget || 0).toLocaleString()}
อนุมัติ: ${Number(item.Approved_Budget || 0).toLocaleString()}
${item.Operation_Status} ${item.Payment_Status} `).join(''); }
